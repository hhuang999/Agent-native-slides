04 / 05 — ARCHITECTURE

NexOS 3.0 is a 2.4B parameter sparse model compiled to a custom WASM target, running natively on consumer NPUs.

nexos_kernel.rs
// NexOS on-device inference pipeline
pub struct InferenceEngine {
  model: SparseModel,
  vault: HardwareVault,
  npu: NpuContext,
}

impl InferenceEngine {
  pub fn run(&self, prompt: &str) -> TokenStream {
    // Tokenize entirely on-chip
    let tokens = self.model.tokenize(prompt);
    // Route to active expert subset only
    let experts = self.model.route(&tokens);
    // NPU dispatch — never touches main RAM
    self.npu.dispatch(experts)
  }
}

// Zero-copy vault: AES-256 in secure enclave
pub fn seal(data: &[u8]) -> SealedBlob {
  HardwareVault::encrypt_inplace(data, AES256GCM)
}
